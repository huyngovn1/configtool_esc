cmake_minimum_required(VERSION 3.16)
project(BasicEscConfig VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTOUIC ON)
find_package(QT NAMES Qt6 Qt5 REQUIRED COMPONENTS Core Widgets SerialPort)
find_package(Qt${QT_VERSION_MAJOR} REQUIRED COMPONENTS Core Widgets SerialPort)
if(MSVC)
    add_compile_options(/utf-8)
endif()
add_library(esc_core STATIC
    src/protocol/BF_ROOTLOADER.cpp src/protocol/BF_ROOTLOADER.h
    src/protocol/fourwayif.cpp src/protocol/fourwayif.h
    src/model/settingsmodel.cpp src/model/settingsmodel.h)
target_include_directories(esc_core PUBLIC src src/protocol src/model)
target_link_libraries(esc_core PUBLIC Qt${QT_VERSION_MAJOR}::Core)
add_library(esc_ui STATIC
    src/mainwindow.cpp src/mainwindow.h src/mainwindow.ui
    src/backend/escsession.cpp src/backend/escsession.h
    src/backend/serialframes.h src/backend/memoryaddress.h)
target_link_libraries(esc_ui PUBLIC esc_core Qt${QT_VERSION_MAJOR}::Widgets Qt${QT_VERSION_MAJOR}::SerialPort)
add_executable(BasicEscConfig WIN32 src/main.cpp)
target_link_libraries(BasicEscConfig PRIVATE esc_ui)
include(CTest)
if(BUILD_TESTING)
    add_executable(protocol_tests tests/protocol_tests.cpp)
    target_link_libraries(protocol_tests PRIVATE esc_core)
    add_test(NAME protocol COMMAND protocol_tests)
    add_executable(settings_tests tests/settings_tests.cpp)
    target_link_libraries(settings_tests PRIVATE esc_core)
    add_test(NAME settings COMMAND settings_tests)
    add_executable(backend_tests tests/backend_tests.cpp)
    target_link_libraries(backend_tests PRIVATE esc_ui)
    add_test(NAME backend COMMAND backend_tests)
    add_executable(ui_smoke tests/ui_smoke.cpp)
    target_link_libraries(ui_smoke PRIVATE esc_ui)
    add_test(NAME ui_smoke COMMAND ui_smoke)
    set_tests_properties(ui_smoke PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
